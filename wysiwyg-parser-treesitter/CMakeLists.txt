# Android-safe CMake config for the tree-sitter-markdown + tree-sitter-markdown-inline grammars.
#
# The ktreesitter-plugin's generated CMakeLists.txt uses find_package(JNI REQUIRED), which
# fails under AGP's externalNativeBuild — the NDK toolchain doesn't satisfy FindJNI's
# desktop-JDK expectations. This file bypasses that by relying on the NDK's sysroot for
# jni.h. Tracking: https://github.com/tree-sitter/kotlin-tree-sitter/issues/60.
#
# Both grammars compile into the same .so. Markdown needs two grammars because CommonMark
# parses in two phases — block structure first, then inline content inside each block. The
# inline parser gets the block pass's inline byte ranges via Parser.includedRanges at
# runtime, so both Language objects must be loadable from a single native library.
#
# Scanner external symbols are namespaced per grammar (tree_sitter_markdown_* vs
# tree_sitter_markdown_inline_*), so linking both pairs into one .so has no collisions.
#
# GRAMMAR_DIR, GRAMMAR_INLINE_DIR, and GENERATED_DIR are passed in from the module's
# build.gradle via externalNativeBuild.cmake.arguments.

cmake_minimum_required(VERSION 3.22.1)
project(ktreesitter-markdown LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_VISIBILITY_PRESET hidden)
add_compile_options(-Wall -Wextra -Wno-unused-parameter -Werror=implicit-function-declaration)
add_compile_definitions(TREE_SITTER_HIDE_SYMBOLS)

if(NOT DEFINED GRAMMAR_DIR OR NOT DEFINED GRAMMAR_INLINE_DIR OR NOT DEFINED GENERATED_DIR OR NOT DEFINED TREE_SITTER_DIR)
  message(FATAL_ERROR "GRAMMAR_DIR, GRAMMAR_INLINE_DIR, GENERATED_DIR, and TREE_SITTER_DIR must be passed via -D")
endif()

include_directories(
  ${GRAMMAR_DIR}/bindings/c
  ${GRAMMAR_INLINE_DIR}/bindings/c
  ${TREE_SITTER_DIR}/lib/include
  ${TREE_SITTER_DIR}/lib/src
)

# binding.c is generated by ktreesitter-plugin for the block grammar; binding_inline.c is
# a hand-written mirror for the inline grammar (the plugin's Gradle DSL only supports one
# grammar per module, so the second JNI export is 10 hand-written lines).
#
# utf16_parse.c + tree-sitter core (lib.c amalgamates all core sources) give us our own
# ts_parser_parse_string_encoding entry point since ktreesitter hardcodes UTF-8. See utf16_parse.c.
add_library(ktreesitter-markdown SHARED
  ${GENERATED_DIR}/src/jni/binding.c
  ${CMAKE_CURRENT_SOURCE_DIR}/jni/binding_inline.c
  ${CMAKE_CURRENT_SOURCE_DIR}/jni/utf16_parse.c
  ${GRAMMAR_DIR}/src/parser.c
  ${GRAMMAR_DIR}/src/scanner.c
  ${GRAMMAR_INLINE_DIR}/src/parser.c
  ${GRAMMAR_INLINE_DIR}/src/scanner.c
  ${TREE_SITTER_DIR}/lib/src/lib.c
)

# Force 16 KB page alignment in the ELF LOAD segments, required for Android devices using
# 16 KB memory pages (Pixel 8+ and other newer ARM hardware running Android 15+). This whole
# file can go away once https://github.com/tree-sitter/kotlin-tree-sitter/issues/60 is fixed
# and the plugin's generated CMakeLists.txt can be used directly.
target_link_options(ktreesitter-markdown PRIVATE "-Wl,-z,max-page-size=16384")

set_target_properties(ktreesitter-markdown PROPERTIES DEFINE_SYMBOL "")
